// Auto-fill by name (400ms debounce)
useEffect(() => {
  const timer = setTimeout(async () => {
    if (form.fullName.trim().length < 2) return;
    const { data } = await supabase
      .from('persons')
      .select('id, full_name, identification_number')
      .ilike('full_name', `%${form.fullName.trim()}%`)
      .eq('person_category', 'employee');
    if (data && data.length > 0) {
      setForm(prev => ({
        ...prev,
        category: data[0].person_category,
        idNumber: data[0].identification_number
      }));
    } // Auto-filled!
  });
})

Database Schema

📋 Core Tables

TableDescription

vehiclesVehicle registrations
gate_movementsEntry/exit records
audit_logsAction audit trail

🔐 RLS Policies

Row Level Security policies control data access:

  • • Administrator: Full access to all tables
  • • Security Officer: Read/insert to visitors, gate_movements
  • • Management: Read access to dashboard data
  • • Viewer: Read-only access
  • • SECURITY DEFINER functions: is_admin(), is_security_officer()

Security & Compliance

No local dependencies, secure cloud architecture

🔒 Security Features

  • Service role key never exposed client-side
  • Server-side API routes with auth
  • Row Level Security (RLS) policies
  • HTTPS enforced (Vercel)
  • Supabase authentication
  • SQL injection prevention (Supabase)

☁️ Zero Local Dependencies

  • No localhost or file:// references
  • All backend on Supabase cloud
  • All API calls to Vercel/CDN
  • Environment variables on Vercel
  • Production SSL/TLS certificates
  • Cookie-based session management
// Server-side API route
export async GET(request) {
  const { supabase } = createSupabaseServerClient(request);
  const { data: { user } } = await supabase.auth.getUser();
  if (!user) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  const { data } = await supabase.from('persons').select('*');
  return NextResponse.json({ data });
